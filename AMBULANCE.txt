/*
  Traffic System with RFID & IR Detection
  - RFID checks only during RED/YELLOW light
  - Ambulance (B8:47:01:12) gets green light with no buzzer
  - Other vehicles (62:85:8B:51) trigger buzzer alarm
  - IR sensor detects vehicle presence
  
  Connections:
  - Arduino UNO
  - RC522 MFRC522 on SPI:
      SDA(SS) -> D10
      SCK -> D13
      MOSI -> D11
      MISO -> D12
      RST -> D9
      VCC -> 3.3V
      GND -> GND
  - IR OUT -> D6
  - LEDs: RED D2, YELLOW D3, GREEN D4 (with 220Ω resistors)
  - Buzzer -> D5
*/

#include <SPI.h>
#include <MFRC522.h>

/* -------------------- CONFIGURATION -------------------- */

// Pin Definitions
const uint8_t PIN_LED_RED = 2;
const uint8_t PIN_LED_YELLOW = 3;
const uint8_t PIN_LED_GREEN = 4;
const uint8_t PIN_BUZZER = 5;
const uint8_t PIN_IR_SENSOR = 6;

// RFID Pins
const uint8_t SS_PIN = 10;   // SDA pin
const uint8_t RST_PIN = 9;

// Buzzer Type Configuration
const bool PASSIVE_BUZZER = false;  // true for passive, false for active

// Timing Configuration (in milliseconds)
const unsigned long GREEN_TIME = 8000;     // Green light duration
const unsigned long YELLOW_TIME = 2000;    // Yellow light duration
const unsigned long RED_TIME = 8000;       // Red light duration
const unsigned long AMBULANCE_MAX_MS = 30000;  // Max override time for ambulance
const unsigned long RFID_READ_TIMEOUT = 2000;  // Time to wait for RFID tag
const unsigned long ALARM_DURATION = 4000;     // Buzzer alarm duration
const unsigned long ALARM_FLASH_MS = 200;      // Red LED flash interval
const unsigned long IR_DEBOUNCE_MS = 50;       // IR debounce time

// RFID UIDs - Update these with your actual tag UIDs
// Ambulance Tag: B8:47:01:12
const uint8_t ambulanceUID[] = { 0xB8, 0x47, 0x01, 0x12 };
const size_t ambulanceUID_len = 4;

// Normal Vehicle Tag: 62:85:8B:51
const uint8_t normalUID[] = { 0x62, 0x85, 0x8B, 0x51 };
const size_t normalUID_len = 4;

/* -------------------- GLOBAL VARIABLES -------------------- */

// Traffic Light States
enum LightState {
  STATE_RED,
  STATE_YELLOW,
  STATE_GREEN,
  STATE_AMBULANCE_OVERRIDE
};

LightState currentState = STATE_RED;
unsigned long stateStartMillis = 0;

// Ambulance State
bool ambulanceActive = false;
unsigned long ambulanceStartMillis = 0;

// IR Sensor State
bool lastIRState = HIGH;  // Assuming IR sensor gives HIGH when no obstacle
bool irDetected = false;
unsigned long lastIRChangeTime = 0;

// Alarm State
bool alarmActive = false;
unsigned long alarmStartMillis = 0;
bool alarmLEDState = false;
unsigned long lastFlashTime = 0;

// RFID Reader
MFRC522 mfrc522(SS_PIN, RST_PIN);

/* -------------------- HELPER FUNCTIONS -------------------- */

// Convert byte array to string for printing
String uidToString(byte *buffer, byte bufferSize) {
  String uidString = "";
  for (byte i = 0; i < bufferSize; i++) {
    if (buffer[i] < 0x10) uidString += "0";
    uidString += String(buffer[i], HEX);
    if (i < bufferSize - 1) uidString += ":";
  }
  uidString.toUpperCase();
  return uidString;
}

// Compare two UIDs
bool compareUID(byte *uid1, byte *uid2, byte size) {
  for (byte i = 0; i < size; i++) {
    if (uid1[i] != uid2[i]) {
      return false;
    }
  }
  return true;
}

// Update LED outputs based on current state
void updateLights() {
  // Don't update lights during alarm (alarm handles red flashing)
  if (alarmActive) return;
  
  digitalWrite(PIN_LED_RED, LOW);
  digitalWrite(PIN_LED_YELLOW, LOW);
  digitalWrite(PIN_LED_GREEN, LOW);
  
  switch (currentState) {
    case STATE_RED:
      digitalWrite(PIN_LED_RED, HIGH);
      break;
    case STATE_YELLOW:
      digitalWrite(PIN_LED_YELLOW, HIGH);
      break;
    case STATE_GREEN:
    case STATE_AMBULANCE_OVERRIDE:
      digitalWrite(PIN_LED_GREEN, HIGH);
      break;
  }
}

// Change traffic light state
void changeState(LightState newState) {
  if (currentState != newState) {
    currentState = newState;
    stateStartMillis = millis();
    updateLights();
    
    Serial.print("State changed to: ");
    switch (newState) {
      case STATE_RED: Serial.println("RED"); break;
      case STATE_YELLOW: Serial.println("YELLOW"); break;
      case STATE_GREEN: Serial.println("GREEN"); break;
      case STATE_AMBULANCE_OVERRIDE: Serial.println("AMBULANCE OVERRIDE"); break;
    }
  }
}

// Start buzzer alarm
void startAlarm() {
  if (!alarmActive) {
    alarmActive = true;
    alarmStartMillis = millis();
    alarmLEDState = false;
    lastFlashTime = 0;
    
    Serial.println("ALARM: Normal vehicle detected during RED/YELLOW!");
    
    // Turn off all LEDs initially
    digitalWrite(PIN_LED_RED, LOW);
    digitalWrite(PIN_LED_YELLOW, LOW);
    digitalWrite(PIN_LED_GREEN, LOW);
  }
}

// Stop buzzer alarm
void stopAlarm() {
  if (alarmActive) {
    alarmActive = false;
    
    // Turn off buzzer
    if (PASSIVE_BUZZER) {
      noTone(PIN_BUZZER);
    } else {
      digitalWrite(PIN_BUZZER, LOW);
    }
    
    // Restore normal lights
    updateLights();
    Serial.println("ALARM: Stopped");
  }
}

// Handle alarm flashing and buzzer
void handleAlarm() {
  if (!alarmActive) return;
  
  unsigned long currentMillis = millis();
  
  // Check if alarm duration has expired
  if (currentMillis - alarmStartMillis >= ALARM_DURATION) {
    stopAlarm();
    return;
  }
  
  // Flash red LED
  if (currentMillis - lastFlashTime >= ALARM_FLASH_MS) {
    lastFlashTime = currentMillis;
    alarmLEDState = !alarmLEDState;
    digitalWrite(PIN_LED_RED, alarmLEDState ? HIGH : LOW);
    
    // Control buzzer
    if (alarmLEDState) {
      if (PASSIVE_BUZZER) {
        tone(PIN_BUZZER, 2000); // 2kHz tone for passive buzzer
      } else {
        digitalWrite(PIN_BUZZER, HIGH); // Active buzzer ON
      }
    } else {
      if (PASSIVE_BUZZER) {
        noTone(PIN_BUZZER);
      } else {
        digitalWrite(PIN_BUZZER, LOW); // Active buzzer OFF
      }
    }
  }
}

// Read RFID tag with timeout
bool readRFIDTag(byte *readUID, byte &uidSize) {
  unsigned long startTime = millis();
  
  while (millis() - startTime < RFID_READ_TIMEOUT) {
    // Check for new RFID card
    if (mfrc522.PICC_IsNewCardPresent() && mfrc522.PICC_ReadCardSerial()) {
      // Copy UID
      uidSize = mfrc522.uid.size;
      for (byte i = 0; i < uidSize; i++) {
        readUID[i] = mfrc522.uid.uidByte[i];
      }
      
      // Print detected UID
      Serial.print("RFID Detected: ");
      Serial.println(uidToString(readUID, uidSize));
      
      // Halt PICC and stop crypto
      mfrc522.PICC_HaltA();
      mfrc522.PCD_StopCrypto1();
      
      return true;
    }
    delay(50); // Small delay to prevent bus congestion
  }
  
  Serial.println("RFID: No tag detected within timeout");
  return false;
}

/* -------------------- SETUP -------------------- */
void setup() {
  // Initialize Serial
  Serial.begin(9600);
  while (!Serial); // Wait for serial port to connect (for Leonardo/Micro)
  Serial.println("\nTraffic System with RFID & IR - Initializing...");
  
  // Initialize pins
  pinMode(PIN_LED_RED, OUTPUT);
  pinMode(PIN_LED_YELLOW, OUTPUT);
  pinMode(PIN_LED_GREEN, OUTPUT);
  pinMode(PIN_BUZZER, OUTPUT);
  pinMode(PIN_IR_SENSOR, INPUT_PULLUP); // Using internal pull-up
  
  // Ensure all outputs are off initially
  digitalWrite(PIN_LED_RED, LOW);
  digitalWrite(PIN_LED_YELLOW, LOW);
  digitalWrite(PIN_LED_GREEN, LOW);
  digitalWrite(PIN_BUZZER, LOW);
  
  // Initialize SPI and RFID reader
  SPI.begin();
  mfrc522.PCD_Init();
  delay(100); // Allow time for RC522 to initialize
  
  // Check RC522 version
  byte version = mfrc522.PCD_ReadRegister(mfrc522.VersionReg);
  Serial.print("MFRC522 Version: 0x");
  Serial.println(version, HEX);
  
  if (version == 0x00 || version == 0xFF) {
    Serial.println("ERROR: MFRC522 not found. Check wiring!");
    while (true); // Halt if RFID not detected
  }
  
  Serial.println("MFRC522 Ready!");
  
  // Initial state
  changeState(STATE_RED);
  stateStartMillis = millis();
  
  // Initial IR state
  lastIRState = digitalRead(PIN_IR_SENSOR);
  
  Serial.println("System Ready!");
  Serial.println("==================================");
}

/* -------------------- MAIN LOOP -------------------- */
void loop() {
  unsigned long currentMillis = millis();
  
  // Handle alarm if active
  handleAlarm();
  
  // Read IR sensor with debouncing
  bool currentIRState = digitalRead(PIN_IR_SENSOR);
  
  // Check for IR state change (vehicle detection)
  if (currentIRState != lastIRState) {
    if (currentMillis - lastIRChangeTime > IR_DEBOUNCE_MS) {
      // Valid state change detected
      if (currentIRState == LOW) {  // Assuming LOW means obstacle detected
        Serial.println("IR: Vehicle detected!");
        irDetected = true;
        
        // Only process RFID during RED or YELLOW (not during GREEN or ambulance override)
        if (!ambulanceActive && !alarmActive) {
          if (currentState == STATE_RED || currentState == STATE_YELLOW) {
            Serial.println("Processing RFID for vehicle at RED/YELLOW light...");
            
            // Read RFID tag
            byte readUID[10];
            byte uidSize = 0;
            
            if (readRFIDTag(readUID, uidSize)) {
              // Check if it's an ambulance
              if (uidSize == ambulanceUID_len && compareUID(readUID, (byte*)ambulanceUID, uidSize)) {
                Serial.println("AMBULANCE DETECTED! Granting override...");
                
                // Activate ambulance override
                ambulanceActive = true;
                ambulanceStartMillis = currentMillis;
                changeState(STATE_AMBULANCE_OVERRIDE);
              }
              // Check if it's a normal vehicle
              else if (uidSize == normalUID_len && compareUID(readUID, (byte*)normalUID, uidSize)) {
                Serial.println("NORMAL VEHICLE DETECTED during RED/YELLOW!");
                startAlarm();
              }
              else {
                Serial.println("UNKNOWN VEHICLE DETECTED!");
                startAlarm();
              }
            } else {
              // No RFID tag detected - treat as normal vehicle
              Serial.println("No RFID tag presented during RED/YELLOW!");
              startAlarm();
            }
          }
          else if (currentState == STATE_GREEN) {
            Serial.println("Vehicle detected at GREEN light - No RFID check needed");
          }
        }
        else if (ambulanceActive) {
          Serial.println("Ambulance already has override - vehicle ignored");
        }
        else if (alarmActive) {
          Serial.println("Alarm active - vehicle ignored");
        }
      }
      else {
        // IR returned to HIGH (no obstacle)
        Serial.println("IR: Vehicle passed");
        irDetected = false;
        
        // If ambulance was active and vehicle passed, end override
        if (ambulanceActive) {
          Serial.println("Ambulance has passed - ending override");
          ambulanceActive = false;
          // Return to RED state for safety
          changeState(STATE_RED);
        }
      }
      
      lastIRChangeTime = currentMillis;
      lastIRState = currentIRState;
    }
  }
  
  // Check ambulance timeout
  if (ambulanceActive && (currentMillis - ambulanceStartMillis >= AMBULANCE_MAX_MS)) {
    Serial.println("Ambulance override timeout - returning to normal");
    ambulanceActive = false;
    changeState(STATE_RED);
  }
  
  // Normal traffic light cycle (only when no ambulance and no alarm)
  if (!ambulanceActive && !alarmActive) {
    unsigned long stateTime = currentMillis - stateStartMillis;
    
    switch (currentState) {
      case STATE_GREEN:
        if (stateTime >= GREEN_TIME) {
          changeState(STATE_YELLOW);
        }
        break;
        
      case STATE_YELLOW:
        if (stateTime >= YELLOW_TIME) {
          changeState(STATE_RED);
        }
        break;
        
      case STATE_RED:
        if (stateTime >= RED_TIME) {
          changeState(STATE_GREEN);
        }
        break;
        
      case STATE_AMBULANCE_OVERRIDE:
        // This should not happen here as ambulanceActive would be true
        break;
    }
  }
  
  // Small delay to prevent watchdog issues
  delay(10);
}
